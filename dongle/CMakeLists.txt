cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(apex_receiver)
add_subdirectory(../radio ${CMAKE_CURRENT_BINARY_DIR}/apex_radio)
target_sources(app PRIVATE src/main.c src/usb.c src/crypto_test.c ../radio/src/apex_aes_ecb.c)
target_sources(app PRIVATE src/pair_store.c ../radio/src/apex_pair_shell.c)
if(CONFIG_APEX_RECEIVER_RADIO_PROBE)
  target_sources(app PRIVATE ../radio/src/apex_radio_probe.c)
endif()
if(CONFIG_APEX_RECEIVER_RADIO_INPUT)
  target_sources(app PRIVATE src/hid.c ../radio/src/apex_input.c)
endif()
target_link_options(zephyr_interface INTERFACE "-Wl,--wrap=tc_aes_encrypt")
if(DEFINED APEX_RECEIVER_BUILD_ID)
  target_compile_definitions(app PRIVATE APEX_RECEIVER_BUILD_ID="${APEX_RECEIVER_BUILD_ID}")
endif()
